# Development dependencies for the academic-search MCP server.
# Install alongside requirements.txt:
#
#   python -m venv .venv
#   .venv\Scripts\activate            # Windows
#   pip install -r requirements.txt -r requirements-dev.txt
#
# Or in one shot with uv:
#
#   uv pip install --system -r requirements.txt -r requirements-dev.txt
#
# Or fully ephemeral with uv (no venv required):
#
#   uv run --with-requirements requirements.txt --with-requirements requirements-dev.txt pytest
